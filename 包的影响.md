### 1. @szew/icons-generic-webfont

**引用位置：** `packages/modules/viewer-react/src/components/iWork/IWorkLoader.tsx`

```ts
import "@szew/icons-generic-webfont/dist/szew-icons-generic-webfont.css";
```

**使用模式：**

1. 导入 → 全局注册 `@font-face` 和 CSS class
2. iProject UI 组件内部渲染类似 `<i class="szew-icon-settings">` 的元素
3. 浏览器自动匹配 CSS class 显示对应图标

源码里不会出现 `icons-generic-webfont` 的显式调用——它通过 CSS class 名隐式匹配，真正的"使用方"是 `@iproject/appui-react`、`@iproject/core-react` 等组件运行时生成的 DOM。

删掉后图标正常显示，说明这些外部包已经自己带了这个依赖，不再需要通过 `viewer-react` 间接提供。

## 三、ECSchema 与 C++ / TS 类映射

> 参考：https://github.com/iProject/bis-schemas

### 核心机制

C++ struct 名称和 EC schema 中 class 名称**严格对应**，映射是显式做的，不是靠名字巧合。

以 `GeometricModel3d` 为例：

**EC schema 侧 (BisCore.ecschema.xml)：**
```xml
<ECEntityClass typeName="GeometricModel3d" ...>
```

**C++ 侧 (DgnModel.h:954-956)：**
```cpp
struct GeometricModel3d : GeometricModel {
    DGNMODEL_DECLARE_MEMBERS(BIS_CLASS_GeometricModel3d, GeometricModel);
};
```

`BIS_CLASS_GeometricModel3d` 是在 `DgnDbTables.h:59` 定义的字符串常量：
```cpp
#define BIS_CLASS_GeometricModel3d "GeometricModel3d"
```

`DGNMODEL_DECLARE_MEMBERS` 宏展开后注册了这个字符串（`DgnModel.h:39-43`）：
```cpp
static Utf8CP MyHandlerECClassName() { return "GeometricModel3d"; }
Utf8CP _GetHandlerECClassName() const override { return MyHandlerECClassName(); }
```

### 继承层次对照

```
EC Schema (XML)                              C++ (DgnModel.h)
─────────────────                            ─────────────────
DgnModel          ───────────────────────▶   DgnModel
  └─ GeometricModel  ──────────────────▶       └─ GeometricModel   (abstract, DEFINE_T_SUPER)
       ├─ GeometricModel2d  ──────────▶            ├─ GeometricModel2d
       │    └─ GraphicalModel2d  ─────▶            │    └─ GraphicalModel2d
       └─ GeometricModel3d  ──────────▶            └─ GeometricModel3d
            ├─ PhysicalModel      ────▶                 ├─ ...
            └─ GraphicalModel3d  ─────▶                 └─ GraphicalModel3d
```

运行时通过 `db.Schemas().GetClassId("BisCore", "GeometricModel")` 这种查表方式，用字符串名找到对应 classId，再路由到正确的 C++ handler。

### C++ 实现 [DgnModel.h]

```cpp
struct EXPORT_VTABLE_ATTRIBUTE GeometricModel3d : GeometricModel
{
    DGNMODEL_DECLARE_MEMBERS(BIS_CLASS_GeometricModel3d, GeometricModel);
    friend struct dgn_ModelHandler::Geometric3d;

protected:
    BE_JSON_NAME(isNotSpatiallyLocated)
    BE_JSON_NAME(isPlanProjection)

    BE_PROP_NAME(IsNotSpatiallyLocated)
    BE_PROP_NAME(IsPlanProjection)

    bool m_isNotSpatiallyLocated = false;
    bool m_isPlanProjection = false;

    DGNPLATFORM_EXPORT DgnDbStatus _FillRangeIndex() override;
    DGNPLATFORM_EXPORT AxisAlignedBox3d _QueryElementsRange() const override;
    GeometricModel3dCP _ToGeometricModel3d() const override final { return this; }
    DGNPLATFORM_EXPORT DgnDbStatus _OnInsertElement(DgnElementR element) override;
    DGNPLATFORM_EXPORT void _BindWriteParams(BeSQLite::EC::ECSqlStatement& statement,
        ForInsert forInsert) override;
    DGNPLATFORM_EXPORT DgnDbStatus _ReadSelectParams(BeSQLite::EC::ECSqlStatement&,
        ECSqlClassParamsCR) override;
    DGNPLATFORM_EXPORT void _ToJson(BeJsValue out, BeJsConst opts) const override;
    DGNPLATFORM_EXPORT void _FromJson(BeJsConst props) override;
    explicit GeometricModel3d(CreateParams const& params) : T_Super(params) {}
    DGNPLATFORM_EXPORT AxisAlignedBox3d _GetElementRange(DgnElementId id) const override;

public:
    //! Test whether this GeometricModel3d is spatially located.
    //! Note that template models are not in the spatial index.
    bool IsSpatiallyLocated() const { return !IsNotSpatiallyLocated(); }
    bool IsNotSpatiallyLocated() const { return m_isNotSpatiallyLocated || IsTemplate(); }
    void SetNotSpatiallyLocated() { m_isNotSpatiallyLocated = true; }
    // If true, then the elements in this GeometricModel3d are expected to be in an XY plane.
    bool IsPlanProjection() const { return m_isPlanProjection; }
    void SetIsPlanProjection(bool isPlanProjection) { m_isPlanProjection = isPlanProjection; }

    bool Is3d() const final { return true; }
};
```

### TypeScript 实现 [Model.ts]

```ts
export abstract class GeometricModel3d extends GeometricModel {
  /** If true, then the elements in this GeometricModel3d are expected to be in an XY plane.
   * @note The associated ECProperty was added to the BisCore schema in version 1.0.8
   */
  public readonly isPlanProjection: boolean;

  /** If true, then the elements in this GeometricModel3d are not in real-world coordinates
   * and will not be in the spatial index.
   * @note The associated ECProperty was added to the BisCore schema in version 1.0.8
   */
  public readonly isNotSpatiallyLocated: boolean;

  /** If true, then the elements in this GeometricModel3d are in real-world coordinates
   * and will be in the spatial index. */
  public get isSpatiallyLocated(): boolean { return !this.isNotSpatiallyLocated; }

  public static override get className(): string { return "GeometricModel3d"; }

  protected constructor(props: GeometricModel3dProps, iWork: IWorkDb) {
    super(props, iWork);
    this.isNotSpatiallyLocated = JsonUtils.asBool(props.isNotSpatiallyLocated);
    this.isPlanProjection = JsonUtils.asBool(props.isPlanProjection);
  }

  /** @inheritdoc @beta */
  protected static override readonly _customHandledProps: CustomHandledProperty[] = [
    { propertyName: "isPlanProjection", source: "Class" },
    { propertyName: "isNotSpatiallyLocated", source: "Class" },
  ];

  /** @inheritdoc @beta */
  public static override deserialize(props: DeserializeEntityArgs): GeometricModel3dProps {
    const modelProps = super.deserialize(props) as GeometricModel3dProps;
    const instance = props.row;
    if (instance.isNotSpatiallyLocated === true || instance.isTemplate === true)
      modelProps.isNotSpatiallyLocated = true;
    if (instance.isPlanProjection === true)
      modelProps.isPlanProjection = true;
    return modelProps;
  }

  /** @inheritdoc @beta */
  public static override serialize(props: GeometricModel3dProps, _iWork: IWorkDb): ECSqlRow {
    const inst = super.serialize(props, _iWork);
    inst.isNotSpatiallyLocated = props.isNotSpatiallyLocated ?? false;
    inst.isPlanProjection = props.isPlanProjection ?? false;
    return inst;
  }

  public override toJSON(): GeometricModel3dProps {
    const val = super.toJSON() as GeometricModel3dProps;
    if (this.isNotSpatiallyLocated)
      val.isNotSpatiallyLocated = true;
    if (this.isPlanProjection)
      val.isPlanProjection = true;
    return val;
  }
}
```
